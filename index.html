function doPost(e) { try { var ss = SpreadsheetApp.getActiveSpreadsheet(); var data = JSON.parse(e.postData.contents); if (data.action === "delete") { var sheet = ss.getSheetByName(data.area); if (!sheet) sheet = ss.getSheets()[0]; var rows = sheet.getDataRange().getValues(); for (var i = 1; i < rows.length; i++) { if (String(rows[i][0]).trim() === String(data.id).trim()) { sheet.deleteRow(i + 1); break; } } return ContentService.createTextOutput(JSON.stringify({ "result": "deleted" })).setMimeType(ContentService.MimeType.JSON); } var sheet = ss.getSheetByName(data.area); if (!sheet) { sheet = ss.getSheetByName("MEDICINA GENERAL"); if (!sheet) sheet = ss.getSheets()[0]; } var ultimaFila = sheet.getLastRow() + 1; var datosFormulario = [[ data.id, data.fecha, data.horaInicio, data.horaFinal, data.nombres, data.dni, data.diagnostico, data.medicamentos, data.medico, data.observaciones ]]; sheet.getRange(ultimaFila, 1, 1, datosFormulario[0].length).setValues(datosFormulario); return ContentService.createTextOutput(JSON.stringify({ "result": "success" })).setMimeType(ContentService.MimeType.JSON); } catch (error) { return ContentService.createTextOutput(JSON.stringify({ "result": "error", "error": error.toString() })).setMimeType(ContentService.MimeType.JSON); } } // FUNCIÓN DEFINITIVA PARA EVITAR BLOQUEOS DEL NAVEGADOR function doGet(e) { var params = e && e.parameter ? e.parameter : {}; var callback = params.callback; // Recibe la petición JSONP if (params.action === "login") { var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheetUsuarios = ss.getSheetByName("USUARIOS"); var acceso = false; if (sheetUsuarios) { var rows = sheetUsuarios.getDataRange().getValues(); for (var i = 1; i < rows.length; i++) { var u = String(rows[i][0]).trim().toUpperCase(); var p = String(rows[i][1]).trim(); var a = String(rows[i][2]).trim().toUpperCase(); if (u === String(params.usuario).trim().toUpperCase() && p === String(params.pin).trim() && a === String(params.area).trim().toUpperCase()) { acceso = true; break; } } } var result = { "status": acceso ? "success" : "denied" }; // Si viene por JSONP, devuelve el código empaquetado para saltar la seguridad if (callback) { return ContentService.createTextOutput(callback + '(' + JSON.stringify(result) + ')').setMimeType(ContentService.MimeType.JAVASCRIPT); } else { return ContentService.createTextOutput(JSON.stringify(result)).setMimeType(ContentService.MimeType.JSON); } } return ContentService.createTextOutput(JSON.stringify({status: "activo"})).setMimeType(ContentService.MimeType.JSON); }
